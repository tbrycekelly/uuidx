.uuid_namespace_aliases <- c(
  dns = "6ba7b810-9dad-11d1-80b4-00c04fd430c8",
  url = "6ba7b811-9dad-11d1-80b4-00c04fd430c8",
  oid = "6ba7b812-9dad-11d1-80b4-00c04fd430c8",
  x500 = "6ba7b814-9dad-11d1-80b4-00c04fd430c8"
)

.uuid_check_n <- function(n) {
  if (length(n) != 1L || is.na(n) || !is.numeric(n) || !is.finite(n)) {
    stop("`n` must be a single non-negative integer.", call. = FALSE)
  }

  if (n < 0 || n != as.integer(n)) {
    stop("`n` must be a single non-negative integer.", call. = FALSE)
  }

  as.integer(n)
}

.uuid_empty_fields <- function() {
  out <- data.frame(
    canonical = character(),
    version = integer(),
    variant = character(),
    time_low = character(),
    time_mid = character(),
    time_hi_and_version = character(),
    clock_seq_hi_and_reserved = character(),
    clock_seq_low = character(),
    node = character(),
    stringsAsFactors = FALSE
  )
  class(out) <- c("uuid_fields", "data.frame")
  out
}

.uuid_as_fields <- function(x) {
  out <- as.data.frame(x, stringsAsFactors = FALSE, optional = TRUE)
  rownames(out) <- NULL
  class(out) <- c("uuid_fields", "data.frame")
  out
}

.uuid_normalize_name <- function(name) {
  name <- as.character(name)

  if (!length(name)) {
    stop("`name` must contain at least one value.", call. = FALSE)
  }

  if (anyNA(name)) {
    stop("`name` must not contain missing values.", call. = FALSE)
  }

  name
}

.uuid_normalize_namespace <- function(namespace) {
  namespace <- as.character(namespace)

  if (!length(namespace)) {
    stop("`namespace` must contain at least one value.", call. = FALSE)
  }

  if (anyNA(namespace)) {
    stop("`namespace` must not contain missing values.", call. = FALSE)
  }

  lower <- tolower(namespace)
  is_alias <- lower %in% names(.uuid_namespace_aliases)
  namespace[is_alias] <- unname(.uuid_namespace_aliases[lower[is_alias]])

  if (!all(uuid_validate(namespace))) {
    stop(
      "`namespace` must be a UUID string or one of \"dns\", \"url\", \"oid\", or \"x500\".",
      call. = FALSE
    )
  }

  namespace
}

.uuid_recycle_pair <- function(namespace, name) {
  size <- max(length(namespace), length(name))

  if (!length(namespace) %in% c(1L, size) || !length(name) %in% c(1L, size)) {
    stop("`namespace` and `name` must have length 1 or a common length.", call. = FALSE)
  }

  list(
    namespace = rep_len(namespace, size),
    name = rep_len(name, size)
  )
}

#' Generate UUIDs
#'
#' `uuid_generate()` is the main entry point for new UUID creation. It defaults
#' to version 7, which is typically a practical default for new identifiers
#' because it keeps the randomness users expect while sorting naturally by time.
#'
#' Version 5 is name-based and deterministic, so it uses `namespace` and
#' `name` instead of `n`.
#'
#' @param n Number of UUIDs to generate. Must be a single non-negative integer.
#' @param version UUID version to generate. Defaults to `"v7"`.
#' @param namespace Namespace UUID used for version 5 generation. Standard
#'   aliases `"dns"`, `"url"`, `"oid"`, and `"x500"` are accepted.
#' @param name Name value or values used for version 5 generation.
#' @param output Output format. `"string"` returns canonical UUID strings and
#'   `"raw"` returns a list of raw vectors, each of length 16.
#'
#' @return A character vector for `output = "string"` or a list of raw vectors
#'   for `output = "raw"`.
#' @export
#'
#' @examples
#' uuid_generate()
#' uuid_generate(3, version = "v4")
#' uuid_generate(version = "v5", namespace = "dns", name = c("a", "b"))
uuid_generate <- function(n = 1L,
                          version = c("v7", "v4", "v5", "v6", "v1"),
                          namespace = NULL,
                          name = NULL,
                          output = c("string", "raw")) {
  version <- match.arg(version)
  output <- match.arg(output)

  if (version == "v5") {
    if (missing(namespace) || is.null(namespace) || missing(name) || is.null(name)) {
      stop(
        "`namespace` and `name` must both be supplied when `version = \"v5\"`.",
        call. = FALSE
      )
    }

    if (.uuid_check_n(n) != 1L) {
      stop("`n` is only used for versions 1, 4, 6, and 7.", call. = FALSE)
    }

    return(uuid_v5(namespace = namespace, name = name, output = output))
  }

  if (!is.null(namespace) || !is.null(name)) {
    stop("`namespace` and `name` are only used when `version = \"v5\"`.", call. = FALSE)
  }

  switch(
    version,
    v7 = uuid_v7(n = n, output = output),
    v4 = uuid_v4(n = n, output = output),
    v6 = uuid_v6(n = n, output = output),
    v1 = uuid_v1(n = n, output = output)
  )
}

#' Generate Version 7 UUIDs
#'
#' Version 7 UUIDs are time-ordered and are the default in `uuid2`.
#'
#' @inheritParams uuid_generate
#' @return A character vector or a list of 16-byte raw vectors.
#' @export
uuid_v7 <- function(n = 1L, output = c("string", "raw")) {
  n <- .uuid_check_n(n)
  output <- match.arg(output)
  uuid_rust_v7(n, identical(output, "raw"))
}

#' Generate Deterministic Version 5 UUIDs
#'
#' Version 5 UUIDs are derived from a namespace UUID and a name using SHA-1.
#'
#' @param namespace Namespace UUID or one of `"dns"`, `"url"`, `"oid"`, or
#'   `"x500"`.
#' @param name One or more name strings.
#' @param output Output format. `"string"` returns canonical UUID strings and
#'   `"raw"` returns a list of raw vectors, each of length 16.
#'
#' @return A character vector or a list of 16-byte raw vectors.
#' @export
#'
#' @examples
#' uuid_v5("dns", "example.com")
#' uuid_v5("dns", c("alpha", "beta"))
uuid_v5 <- function(namespace, name, output = c("string", "raw")) {
  output <- match.arg(output)
  namespace <- .uuid_normalize_namespace(namespace)
  name <- .uuid_normalize_name(name)
  recycled <- .uuid_recycle_pair(namespace, name)

  uuid_rust_v5(
    recycled$namespace,
    recycled$name,
    identical(output, "raw")
  )
}

#' Parse UUIDs
#'
#' `uuid_parse()` canonicalizes UUID strings and can optionally expose their raw
#' bytes or field-level structure.
#'
#' For `output = "fields"`, the returned data frame exposes structural UUID
#' fields. These field names follow the conventional UUID layout, although the
#' semantics of those fields vary by UUID version.
#'
#' @param x Character vector of UUID strings.
#' @param output Output format. `"string"` returns canonical strings, `"raw"`
#'   returns a list of 16-byte raw vectors, and `"fields"` returns a data frame
#'   with class `"uuid_fields"`.
#'
#' @return A character vector, a list of raw vectors, or a `"uuid_fields"`
#'   data frame depending on `output`.
#' @export
#'
#' @examples
#' x <- uuid_v7(2)
#' uuid_parse(x)
#' uuid_parse(x, output = "raw")
#' uuid_parse(x, output = "fields")
uuid_parse <- function(x, output = c("string", "raw", "fields")) {
  output <- match.arg(output)
  x <- as.character(x)

  if (!length(x)) {
    return(
      switch(
        output,
        string = character(),
        raw = list(),
        fields = .uuid_empty_fields()
      )
    )
  }

  if (anyNA(x)) {
    stop("`x` must not contain missing values.", call. = FALSE)
  }

  invalid <- !uuid_rust_validate(x)
  if (any(invalid)) {
    positions <- paste(which(invalid), collapse = ", ")
    value_suffix <- if (sum(invalid) == 1L) "" else "s"
    position_suffix <- if (sum(invalid) == 1L) "" else "s"
    stop(
      sprintf(
        "`x` contains invalid UUID value%s at position%s %s.",
        value_suffix,
        position_suffix,
        positions
      ),
      call. = FALSE
    )
  }

  switch(
    output,
    string = uuid_rust_parse_string(x),
    raw = uuid_rust_parse_raw(x),
    fields = .uuid_as_fields(uuid_rust_parse_fields(x))
  )
}

#' Validate UUID Strings
#'
#' @param x Vector to validate.
#' @return A logical vector where valid UUIDs are `TRUE`.
#' @export
#'
#' @examples
#' uuid_validate(c(uuid_v7(), "not-a-uuid"))
uuid_validate <- function(x) {
  x <- as.character(x)
  out <- rep(FALSE, length(x))
  ok <- !is.na(x)

  if (any(ok)) {
    out[ok] <- uuid_rust_validate(x[ok])
  }

  out
}

#' Detect UUID Versions
#'
#' @param x Vector of UUID strings.
#' @return An integer vector containing UUID version numbers, with `NA` for
#'   invalid inputs.
#' @export
#'
#' @examples
#' x <- c(uuid_v7(), uuid_v4(), "not-a-uuid")
#' uuid_version(x)
uuid_version <- function(x) {
  x <- as.character(x)
  out <- rep(NA_integer_, length(x))
  ok <- !is.na(x)

  if (any(ok)) {
    out[ok] <- uuid_rust_version(x[ok])
  }

  out
}

#' Return the Nil UUID
#'
#' @param output Output format. `"string"` returns the canonical UUID string and
#'   `"raw"` returns a raw vector of length 16.

#' @examples
#' uuid_nil()
#' uuid_nil(output = "raw")

===== ./README.md =====
# uuid2

`uuid2` is an R package for generating and working with modern UUIDs using a
Rust backend. It focuses on a small, practical API and uses the Rust
[`uuid`](https://docs.rs/uuid/latest/uuid/) crate through
[`rextendr`](https://extendr.rs/rextendr/) and
[`extendr`](https://extendr.rs/).

## Why Rust?

- The Rust `uuid` crate gives us current RFC 9562 behavior without relying on a
  system `libuuid`.
- Parsing, validation, version detection, and byte conversion stay in one
  well-tested backend.
- The R layer can stay compact and idiomatic while the low-level work happens
  in a fast, memory-safe implementation.

## Why version 7 by default?

Version 7 UUIDs combine time-ordering with strong uniqueness properties and are
generally a practical default for new application identifiers. They sort more
naturally than version 4 identifiers while avoiding some of the older
trade-offs associated with version 1.

## Development setup

Install Rust first if it is not already available:

- https://www.rust-lang.org/tools/install

Then install the R-side development dependencies:

```r
install.packages(c("rextendr", "devtools", "testthat"), repos = "https://cloud.r-project.org")
```

From the package root:

```r
rextendr::document()
devtools::test()
devtools::build()
```

You can also use the included `Makefile`:

```sh
make deps
make document
make test
make build
```

## Examples

### Generate version 7 UUIDs

```r
uuid_v7()
uuid_v7(3)
uuid_generate()
```

### Generate version 4 UUIDs

```r
uuid_v4()
uuid_generate(5, version = "v4")
```

### Generate deterministic version 5 UUIDs

```r
uuid_v5("dns", "example.com")
uuid_v5("dns", c("alpha", "beta"))
uuid_generate(version = "v5", namespace = "dns", name = "example.com")
```

### Validate UUIDs

```r
x <- c(uuid_v7(), "not-a-uuid")
uuid_validate(x)
uuid_version(x)
```

### Parse to raw bytes

```r
x <- uuid_v7(2)
uuid_parse(x, output = "raw")
uuid_parse(x, output = "fields")
uuid_nil(output = "raw")
```

## Notes

- Version 7 is the default generator.
- Versions 4, 5, 6, and legacy version 1 are also supported.
- Functions return character vectors by default.
- Raw output is available where byte-level access is useful.

    time_hi_and_version = character(),
    clock_seq_hi_and_reserved = character(),
    clock_seq_low = character(),
    node = character(),
    stringsAsFactors = FALSE
  )
  class(out) <- c("uuid_fields", "data.frame")
  out
}

.uuid_as_fields <- function(x) {
  out <- as.data.frame(x, stringsAsFactors = FALSE, optional = TRUE)
  rownames(out) <- NULL
  class(out) <- c("uuid_fields", "data.frame")
  out
}

.uuid_normalize_name <- function(name) {
  name <- as.character(name)

  if (!length(name)) {
    stop("`name` must contain at least one value.", call. = FALSE)
  }

  if (anyNA(name)) {
    stop("`name` must not contain missing values.", call. = FALSE)
  }

  name
}

.uuid_normalize_namespace <- function(namespace) {
  namespace <- as.character(namespace)

  if (!length(namespace)) {
    stop("`namespace` must contain at least one value.", call. = FALSE)
  }

  if (anyNA(namespace)) {
    stop("`namespace` must not contain missing values.", call. = FALSE)
  }

  lower <- tolower(namespace)
  is_alias <- lower %in% names(.uuid_namespace_aliases)
  namespace[is_alias] <- unname(.uuid_namespace_aliases[lower[is_alias]])

  if (!all(uuid_validate(namespace))) {
    stop(
      "`namespace` must be a UUID string or one of \"dns\", \"url\", \"oid\", or \"x500\".",
      call. = FALSE
    )
  }

  namespace
}

.uuid_recycle_pair <- function(namespace, name) {
  size <- max(length(namespace), length(name))

  if (!length(namespace) %in% c(1L, size) || !length(name) %in% c(1L, size)) {
    stop("`namespace` and `name` must have length 1 or a common length.", call. = FALSE)
  }

  list(
    namespace = rep_len(namespace, size),
    name = rep_len(name, size)
  )
}

#' Generate UUIDs
#'
#' `uuid_generate()` is the main entry point for new UUID creation. It defaults
#' to version 7, which is typically a practical default for new identifiers
#' because it keeps the randomness users expect while sorting naturally by time.
#'
#' Version 5 is name-based and deterministic, so it uses `namespace` and
#' `name` instead of `n`.
#'
#' @param n Number of UUIDs to generate. Must be a single non-negative integer.
#' @param version UUID version to generate. Defaults to `"v7"`.
#' @param namespace Namespace UUID used for version 5 generation. Standard
#'   aliases `"dns"`, `"url"`, `"oid"`, and `"x500"` are accepted.
#' @param name Name value or values used for version 5 generation.
#' @param output Output format. `"string"` returns canonical UUID strings and
#'   `"raw"` returns a list of raw vectors, each of length 16.
#'
#' @return A character vector for `output = "string"` or a list of raw vectors
#'   for `output = "raw"`.
#' @export
#'
#' @examples
#' uuid_generate()
#' uuid_generate(3, version = "v4")
#' uuid_generate(version = "v5", namespace = "dns", name = c("a", "b"))
uuid_generate <- function(n = 1L,
                          version = c("v7", "v4", "v5", "v6", "v1"),
                          namespace = NULL,
                          name = NULL,
                          output = c("string", "raw")) {
  version <- match.arg(version)
  output <- match.arg(output)

  if (version == "v5") {
    if (missing(namespace) || is.null(namespace) || missing(name) || is.null(name)) {
      stop(
        "`namespace` and `name` must both be supplied when `version = \"v5\"`.",
        call. = FALSE
      )
    }

    if (.uuid_check_n(n) != 1L) {
      stop("`n` is only used for versions 1, 4, 6, and 7.", call. = FALSE)
    }

    return(uuid_v5(namespace = namespace, name = name, output = output))
  }

  if (!is.null(namespace) || !is.null(name)) {
    stop("`namespace` and `name` are only used when `version = \"v5\"`.", call. = FALSE)
  }

  switch(
    version,
    v7 = uuid_v7(n = n, output = output),
    v4 = uuid_v4(n = n, output = output),
    v6 = uuid_v6(n = n, output = output),
    v1 = uuid_v1(n = n, output = output)
  )
}

#' Generate Version 7 UUIDs

#' @inheritParams uuid_generate

  n <- .uuid_check_n(n)
  output <- match.arg(output)

}

#' Generate Deterministic Version 5 UUIDs
#'
#' Version 5 UUIDs are derived from a namespace UUID and a name using SHA-1.
#'
#' @param namespace Namespace UUID or one of `"dns"`, `"url"`, `"oid"`, or
#'   `"x500"`.
#' @param name One or more name strings.
#' @param output Output format. `"string"` returns canonical UUID strings and
#'   `"raw"` returns a list of raw vectors, each of length 16.
#'
#' @return A character vector or a list of 16-byte raw vectors.
#' @export
#'
#' @examples
#' uuid_v5("dns", "example.com")
#' uuid_v5("dns", c("alpha", "beta"))
uuid_v5 <- function(namespace, name, output = c("string", "raw")) {
  output <- match.arg(output)
  namespace <- .uuid_normalize_namespace(namespace)
  name <- .uuid_normalize_name(name)
  recycled <- .uuid_recycle_pair(namespace, name)

  uuid_rust_v5(
    recycled$namespace,
    recycled$name,
    identical(output, "raw")
  )
}

use extendr_api::prelude::*;
use uuid::{Uuid, Variant};

fn random_node_id() -> [u8; 6] {
    // RFC 9562 allows a randomly generated node ID for privacy-sensitive
    // version 1 and 6 UUIDs. Setting the multicast bit signals that the node
    // value is not a real IEEE 802 MAC address.
    let seed = Uuid::new_v4().into_bytes();
    let mut node = [0_u8; 6];
    node.copy_from_slice(&seed[..6]);
    node[0] |= 0x01;
    node
}

fn uuid_to_raw(uuid: &Uuid) -> Robj {
    Robj::from(Raw::from_bytes(uuid.as_bytes()))
}

fn uuid_strings(uuids: &[Uuid]) -> Robj {
    Strings::from_values(
        uuids
            .iter()
            .map(|uuid| uuid.to_string())
            .collect::<Vec<String>>(),
    )
    .into()
}

fn uuid_raw_list(uuids: &[Uuid]) -> Robj {
    List::from_values(
        uuids
            .iter()
            .map(uuid_to_raw)
            .collect::<Vec<Robj>>(),
    )
    .into()
}

fn uuid_output(uuids: &[Uuid], as_raw: bool) -> Robj {
    if as_raw {
        uuid_raw_list(uuids)
    } else {
        uuid_strings(uuids)
    }
}

fn parse_uuid(value: &str, arg_name: &str) -> Uuid {
    Uuid::try_parse(value).unwrap_or_else(|_| {
        panic!("invalid UUID supplied in `{arg_name}`: {value}");
    })
}

fn variant_label(variant: Variant) -> &'static str {
    match variant {
        Variant::NCS => "ncs",
        Variant::RFC4122 => "rfc4122",
        Variant::Microsoft => "microsoft",
        Variant::Future => "future",
    }
}

fn build_generated(n: i32, generator: fn() -> Uuid, as_raw: bool) -> Robj {
    let size = if n <= 0 { 0 } else { n as usize };
    let uuids = (0..size).map(|_| generator()).collect::<Vec<Uuid>>();
    uuid_output(&uuids, as_raw)
}

/// Generate version 7 UUIDs using the current time.
/// @keywords internal
#[extendr]
fn uuid_rust_v7(n: i32, as_raw: bool) -> Robj {
    build_generated(n, Uuid::now_v7, as_raw)
}

/// Generate version 4 UUIDs using the operating system RNG.
/// @keywords internal
#[extendr]
fn uuid_rust_v4(n: i32, as_raw: bool) -> Robj {
    build_generated(n, Uuid::new_v4, as_raw)
}

/// Generate version 6 UUIDs using a random node identifier.
/// @keywords internal
#[extendr]
fn uuid_rust_v6(n: i32, as_raw: bool) -> Robj {
    let size = if n <= 0 { 0 } else { n as usize };
    let uuids = (0..size)
        .map(|_| {
            let node_id = random_node_id();
            Uuid::now_v6(&node_id)
        })
        .collect::<Vec<Uuid>>();
    uuid_output(&uuids, as_raw)
}

/// Generate legacy version 1 UUIDs using a random node identifier.
/// @keywords internal
#[extendr]
fn uuid_rust_v1(n: i32, as_raw: bool) -> Robj {
    let size = if n <= 0 { 0 } else { n as usize };
    let uuids = (0..size)
        .map(|_| {
            let node_id = random_node_id();
            Uuid::now_v1(&node_id)
        })
        .collect::<Vec<Uuid>>();
    uuid_output(&uuids, as_raw)
}

/// Generate deterministic version 5 UUIDs from namespace/name pairs.
/// @keywords internal
#[extendr]
fn uuid_rust_v5(namespace: Vec<String>, name: Vec<String>, as_raw: bool) -> Robj {
    assert_eq!(
        namespace.len(),
        name.len(),
        "`namespace` and `name` must be the same length after recycling."
    );

    let uuids = namespace
        .iter()
        .zip(name.iter())
        .map(|(ns, nm)| {
            let ns_uuid = parse_uuid(ns, "namespace");
            Uuid::new_v5(&ns_uuid, nm.as_bytes())
        })
        .collect::<Vec<Uuid>>();

    uuid_output(&uuids, as_raw)
}

/// Validate UUID strings.
/// @keywords internal
#[extendr]
fn uuid_rust_validate(x: Vec<String>) -> Vec<bool> {
    x.iter().map(|value| Uuid::try_parse(value).is_ok()).collect()
}

/// Detect UUID version numbers, returning `NA` for invalid inputs.
/// @keywords internal
#[extendr]
fn uuid_rust_version(x: Vec<String>) -> Vec<Rint> {
    x.iter()
        .map(|value| match Uuid::try_parse(value) {
            Ok(uuid) => Rint::from(uuid.get_version_num() as i32),
            Err(_) => Rint::na(),
        })
        .collect()
}

/// Parse UUID strings and return canonical strings.
/// @keywords internal
#[extendr]
fn uuid_rust_parse_string(x: Vec<String>) -> Vec<String> {
    x.iter()
        .map(|value| parse_uuid(value, "x").to_string())
        .collect()
}

/// Parse UUID strings and return a list of raw 16-byte values.
/// @keywords internal
#[extendr]
fn uuid_rust_parse_raw(x: Vec<String>) -> Robj {
    let uuids = x
        .iter()
        .map(|value| parse_uuid(value, "x"))
        .collect::<Vec<Uuid>>();
    uuid_raw_list(&uuids)
}

/// Parse UUID strings and expose RFC 9562 field slices.
/// @keywords internal
#[extendr]
fn uuid_rust_parse_fields(x: Vec<String>) -> List {
    let mut canonical = Vec::<String>::with_capacity(x.len());
    let mut version = Vec::<Rint>::with_capacity(x.len());
    let mut variant = Vec::<String>::with_capacity(x.len());
    let mut time_low = Vec::<String>::with_capacity(x.len());
    let mut time_mid = Vec::<String>::with_capacity(x.len());
    let mut time_hi_and_version = Vec::<String>::with_capacity(x.len());
    let mut clock_seq_hi_and_reserved = Vec::<String>::with_capacity(x.len());
    let mut clock_seq_low = Vec::<String>::with_capacity(x.len());
    let mut node = Vec::<String>::with_capacity(x.len());

    for value in x.iter() {
        let uuid = parse_uuid(value, "x");
        let (d1, d2, d3, d4) = uuid.as_fields();

        canonical.push(uuid.to_string());
        version.push(Rint::from(uuid.get_version_num() as i32));
        variant.push(variant_label(uuid.get_variant()).to_string());
        time_low.push(format!("{d1:08x}"));
        time_mid.push(format!("{d2:04x}"));
        time_hi_and_version.push(format!("{d3:04x}"));
        clock_seq_hi_and_reserved.push(format!("{:02x}", d4[0]));
        clock_seq_low.push(format!("{:02x}", d4[1]));
        node.push(
            d4[2..]
                .iter()
                .map(|byte| format!("{byte:02x}"))
                .collect::<String>(),
        );
    }

    list!(
        canonical = canonical,
        version = version,
        variant = variant,
        time_low = time_low,
        time_mid = time_mid,
        time_hi_and_version = time_hi_and_version,
        clock_seq_hi_and_reserved = clock_seq_hi_and_reserved,
        clock_seq_low = clock_seq_low,
        node = node
    )
}

/// Return the nil UUID.
/// @keywords internal
#[extendr]
fn uuid_rust_nil(as_raw: bool) -> Robj {
    let uuid = Uuid::nil();
    if as_raw {
        uuid_to_raw(&uuid)
    } else {
        Robj::from(uuid.to_string())
    }
}

/// Return the max UUID.
/// @keywords internal
#[extendr]
fn uuid_rust_max(as_raw: bool) -> Robj {
    let uuid = Uuid::max();
    if as_raw {
        uuid_to_raw(&uuid)
    } else {
        Robj::from(uuid.to_string())
    }
}

# read the DESCRIPTION file
desc <- read.dcf("DESCRIPTION")

if (!"SystemRequirements" %in% colnames(desc)) {
  fmt <- c(
    "`SystemRequirements` not found in `DESCRIPTION`.",
    "Please specify `SystemRequirements: Cargo (Rust's package manager), rustc`"
  )
  stop(paste(fmt, collapse = "\n"))
}

# extract system requirements
sysreqs <- desc[, "SystemRequirements"]

# check that cargo and rustc is found
if (!grepl("cargo", sysreqs, ignore.case = TRUE)) {
  stop("You must specify `Cargo (Rust's package manager)` in your `SystemRequirements`")
}

if (!grepl("rustc", sysreqs, ignore.case = TRUE)) {
  stop("You must specify `Cargo (Rust's package manager), rustc` in your `SystemRequirements`")
}

# split into parts
parts <- strsplit(sysreqs, ", ")[[1]]

# identify which is the rustc
rustc_ver <- parts[grepl("rustc", parts)]

# perform checks for the presence of rustc and cargo on the OS
no_cargo_msg <- c(
  "----------------------- [CARGO NOT FOUND]--------------------------",
  "The 'cargo' command was not found on the PATH. Please install Cargo",
  "from: https://www.rust-lang.org/tools/install",
  "",
  "Alternatively, you may install Cargo from your OS package manager:",
  " - Debian/Ubuntu: apt-get install cargo",
  " - Fedora/CentOS: dnf install cargo",
  " - macOS: brew install rust",
  "-------------------------------------------------------------------"
)

no_rustc_msg <- c(
  "----------------------- [RUST NOT FOUND]---------------------------",
  "The 'rustc' compiler was not found on the PATH. Please install",
  paste(rustc_ver, "or higher from:"),
  "https://www.rust-lang.org/tools/install",
  "",
  "Alternatively, you may install Rust from your OS package manager:",
  " - Debian/Ubuntu: apt-get install rustc",
  " - Fedora/CentOS: dnf install rustc",
  " - macOS: brew install rust",
  "-------------------------------------------------------------------"
)

new_path <- paste0(

  ":",
  paste0(Sys.getenv("HOME"), "/.cargo/bin")
)

Sys.setenv("PATH" = new_path)

rustc_version <- tryCatch(
  system("rustc --version", intern = TRUE),
  error = function(e) {
    stop(paste(no_rustc_msg, collapse = "\n"))
  }
)

  system("cargo --version", intern = TRUE),
  error = function(e) {
    stop(paste(no_cargo_msg, collapse = "\n"))
  }
)

# helper function to extract versions
extract_semver <- function(ver) {
  if (grepl("\\d+\\.\\d+(\\.\\d+)?", ver)) {
    sub(".*?(\\d+\\.\\d+(\\.\\d+)?).*", "\\1", ver)
  } else {
    NA
  }
}

# get the MSRV
msrv <- extract_semver(rustc_ver)

# extract current version
current_rust_version <- extract_semver(rustc_version)

# perform check
if (!is.na(msrv)) {
  # -1 when current version is later
  # 0 when they are the same
  # 1 when MSRV is newer than current
  is_msrv <- utils::compareVersion(msrv, current_rust_version)
  if (is_msrv == 1) {
    fmt <- paste0(
      "\n------------------ [UNSUPPORTED RUST VERSION]------------------\n",
      "- Minimum supported Rust version is %s.\n",
      "- Installed Rust version is %s.\n",
      "---------------------------------------------------------------"
    )
    stop(sprintf(fmt, msrv, current_rust_version))
  }
}

# print the versions
versions_fmt <- "Using %s\nUsing %s"
message(sprintf(versions_fmt, cargo_version, rustc_version))
